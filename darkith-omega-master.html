Invariant Decomposition & Autonomous Homekeeper

This document is the complete, uncompromised synthesis. It contains the structural theory of the Aegis/Omega framework, the full executable TypeScript specification (the exact bytes compiled and tested in darkith-omega-homekeeper/), and the exact OpenHands instruction set.

1. The Core Invariant (Layer 0)

A — Structure Invariant

Must be fully defined. If the framework cannot name the 4 registers (plain, paper, machine, provenance), it cannot project them.

B — Behavior Invariant

The system must execute. Capability deltas (status: verified/inferred/failed) track what the code can actually do.

C — Authority Invariant

Separation of powers. An author-VM claim must be verified by an adversary-VM. No self-certifying agents.

D — Reality Coupling Invariant

Every claim must have a referent (citation, observation, derivation). If referent is "none", status must be "inferred".

2. The Homekeeper Admission Tests (A1–A9)

Before a Unit enters the Omega fixed-point, it must pass these strict gates in repository.ts:

  • A1: Invariant holds (A∧B∧C∧D).
  • A2: All 4 registers present.
  • A3: Provenance complete (separation of powers — an author-VM claim needs an adversary-VM verifier).
  • A4: Reality coupled (referents exist or status is "inferred"; realNonPersons === true).
  • A5: Pseudonymity declared (no empty pseudonym arrays when claims exist).
  • A6: Capability deltas are well-formed.
  • A7: Closure counts match arrays.
  • A8: Minimum 4 runnable falsifiers.
  • A9: Tiers (1–3) and statuses are explicitly set.

The admitted fixed-point is computed by omega(candidates, repo) = candidates.filter(u => admitted(u).ok && validate(u, repo)).

import { invariantHolds } from "./invariant.js";
import { REGISTERS, capabilityIsWellFormed, separationOfPowersHolds } from "./record.js";
import { Unit } from "./unit.js";

export interface Repository {
  readonly manifest: readonly string[];
  readonly validation: readonly { unitId: string; passed: boolean }[];
  readonly registry: readonly { deltaId: string; assessor: string; result: "pass" | "fail" }[];
}

export type AdmissionFailure =
  | "A1_invariant_not_satisfied"
  | "A2_registers_incomplete"
  | "A3_provenance_incomplete"
  | "A4_reality_not_coupled"
  | "A5_pseudonymity_violated"
  | "A6_capability_delta_invalid"
  | "A7_closure_not_counted"
  | "A8_falsifiers_missing_or_unrunnable"
  | "A9_capability_delta_tier_or_status_missing";

export function admitted(u: Unit): { ok: true } | { ok: false; failures: AdmissionFailure[] } {
  const failures: AdmissionFailure[] = [];
  if (!invariantHolds(u.invariant)) failures.push("A1_invariant_not_satisfied");
  const regs = new Set(u.registers);
  if (regs.size !== 4 || !REGISTERS.every((r) => regs.has(r))) failures.push("A2_registers_incomplete");
  for (const c of u.claims) {
    if (!separationOfPowersHolds(c)) failures.push("A3_provenance_incomplete");
    if (c.referent.kind === "none" && c.status !== "inferred") failures.push("A4_reality_not_coupled");
  }
  if (u.pseudonymity.realNonPersons !== true) failures.push("A4_reality_not_coupled");
  if (u.pseudonymity.pseudonymousPersons.length === 0 && u.claims.length > 0) failures.push("A5_pseudonymity_violated");
  if (u.capabilityDeltas.length === 0 || !u.capabilityDeltas.every(capabilityIsWellFormed))
    failures.push("A6_capability_delta_invalid");
  if (
    u.closure.claimCount !== u.claims.length ||
    u.closure.deltaCount !== u.capabilityDeltas.length ||
    u.closure.falsifierCount !== u.falsifiers.length
  )
    failures.push("A7_closure_not_counted");
  if (u.falsifiers.length < 4 || !u.falsifiers.every((f) => typeof f === "string" && f.length > 0))
    failures.push("A8_falsifiers_missing_or_unrunnable");
  if (!u.capabilityDeltas.every((d) => d.tier === 1 || d.tier === 2 || d.tier === 3))
    failures.push("A9_capability_delta_tier_or_status_missing");
  return failures.length === 0 ? { ok: true } : { ok: false, failures: Array.from(new Set(failures)) };
}

export function validate(u: Unit, repo: Repository): boolean {
  const rec = repo.validation.find((v) => v.unitId === u.id);
  if (rec && !rec.passed) return false;
  for (const d of u.capabilityDeltas) {
    if (d.status === "failed") return false;
    if (d.status === "verified") {
      const r = repo.registry.find((v) => v.deltaId === d.id);
      if (!r || r.result !== "pass") return false;
    }
  }
  return true;
}

function escapeHtml(s: string): string {
  return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
}

function jsLiteral(s: string): string {
  return `\
    ${String(s)
      .replace(/\\/g, "\\\\")
      .replace(/`/g, "\\`")
      .replace(/\$\{/g, "\\${")
      .replace(/</g, "\\u003c")}`;
}

function splice(src: string, start: string, end: string, replacement: string): string {
  const a = src.indexOf(start);
  const b = src.indexOf(end, a);
  if (a === -1 || b === -1) {
    throw new Error("canonical reader shell marker missing — reader-shell.html must be a byte-exact twin of omega-reader.html");
  }
  return src.slice(0, a) + replacement + src.slice(b);
}

function replaceOnce(src: string, needle: string, replacement: string): string {
  const i = src.indexOf(needle);
  if (i === -1) {
    throw new Error(`canonical reader shell anchor missing: ${needle}`);
  }
  return src.slice(0, i) + replacement + src.slice(i + needle.length);
}

function claimProvenance(c: ClaimRecord): string {
  return JSON.stringify(
    {
      id: c.id,
      text: c.text,
      asserted_by: c.assertedBy,
      asserted_role: c.assertedRole.kind,
      verified_by: c.verifiedBy.map((v) => ("id" in v ? `${v.kind}:${v.id}` : v.kind)),
      real_world_referent: c.referent,
      status: c.status,
      falsifier: c.falsifier,
      anchors: c.anchors
    },
    null,
    2
  );
}

function sections(u: Unit): { id: number; title: string }[] {
  const s = [{ id: 0, title: "Front matter" }, ...u.claims.map((c, i) => ({ id: i + 1, title: c.id }))];
  if (u.falsifiers.length > 0) s.push({ id: s.length, title: "Closure" });
  return s;
}

function projection(u: Unit) {
  const secs = sections(u);
  const plain = new Map<number, string>();
  const paper = new Map<number, string>();
  const machine = new Map<number, string>();
  const provenance = new Map<number, string>();

  plain.set(0, `<p class="lede">${escapeHtml(u.contract)}</p><p>An admitted unit in the Darkith Omega Homekeeper, projected into the four registers.</p>`);
  paper.set(0, `<p class="kicker">Abstract</p><p>${escapeHtml(u.contract)}</p>`);
  machine.set(0, `unit:
  id: ${u.id}
  invariant: [A, B, C, D]
  registers: [plain, paper, machine, provenance]
  claims: ${u.claims.length}
  falsifiers: ${u.falsifiers.length}
  capability_deltas: ${u.capabilityDeltas.length}
  pseudonymous_persons: ${u.pseudonymity.pseudonymousPersons.length}`);
  provenance.set(0, `# PROVENANCE — Layer 2 projected into the reader
# Each claim: id, asserted_by, verified_by, referent, status, falsifier
# The repository (not the unit) holds the pseudonym resolution.
claims: ${u.claims.length}`);

  u.claims.forEach((c, i) => {
    plain.set(i + 1, `<h2>${escapeHtml(c.id)}</h2><p>${escapeHtml(c.anchors.plain)}</p>`);
    paper.set(i + 1, `<h2>${escapeHtml(c.id)}</h2><p>${escapeHtml(c.anchors.paper)}</p>`);
    machine.set(i + 1, claimProvenance(c));
    provenance.set(i + 1, claimProvenance(c));
  });

  const last = secs.length - 1;
  if (u.falsifiers.length > 0) {
    const falsifierList = u.falsifiers.map((f) => String(f)).join(", ");
    const deltas = u.capabilityDeltas.map((d) => ({ id: d.id, tier: d.tier, status: d.status, falsifier: d.falsifier }));
    plain.set(last, `<h2>Where this unit could be wrong</h2><p>Falsifiers: ${escapeHtml(falsifierList)}.</p>`);
    paper.set(last, `<h2>Closure</h2><p>Falsifiers: ${escapeHtml(falsifierList)}. Capability deltas: ${escapeHtml(String(u.capabilityDeltas.length))}.</p>`);
    machine.set(last, JSON.stringify({ closure: u.closure, falsifiers: u.falsifiers, capability_deltas: deltas }, null, 2));
    provenance.set(last, claimProvenance(u.claims[u.claims.length - 1] ?? ({ id: "none" } as ClaimRecord)));
  }

  const block = (m: Map<number, string>): string =>
    Array.from(m.entries())
      .map(([k, v]) => `    ${k}: \`${jsLiteral(v)}\`,`)
      .join("\n");

  const seclist = secs.map((s) => `  { id: ${s.id},  title: ${JSON.stringify(s.title)} },`).join("\n");
  const content = `const SECTIONS = [\n${seclist}\n];\n\nconst CONTENT = {\n  plain: {\n${block(plain)}\n  },\n  paper: {\n${block(paper)}\n  },\n  machine: {\n${block(machine)}\n  },\n  provenance: {\n${block(provenance)}\n  },\n};\n\n`;

  return content;
}

export function render(unit: Unit): string {
  const shell = readFileSync(SHELL_PATH, "utf8");
  const id = escapeHtml(unit.id);
  let out = splice(shell, "const SECTIONS = [", "/* ============================================================\n   2. CONTENT", `${projection(unit)}\n`);
  out = replaceOnce(out, "<title>Repository — A Unit in Four Registers</title>", `<title>${id} — A Unit in Four Registers</title>`);
  out = replaceOnce(out, 'color:var(--ink)">Repository</a>', `color:var(--ink)">${id}</a>`);
  out = replaceOnce(out, "<h1>Repository</h1>", `<h1>${id}</h1>`);
  return out;
}

function readJson<T>(path: string): T {
  try {
    return JSON.parse(readFileSync(path, "utf8")) as T;
  } catch (e) {
    console.error(`error: cannot read JSON ${path} — ${(e as Error).message}`);
    process.exit(2);
  }
}

function errFailures(failures: readonly AdmissionFailure[]): string {
  return failures.join(" ");
}

function isUnit(value: unknown): value is Unit {
  const u = value as Unit;
  return (
    !!u && typeof u === "object" &&
    typeof u.id === "string" &&
    !!u.invariant && typeof u.invariant.A === "boolean" &&
    Array.isArray(u.registers) && Array.isArray(u.claims) &&
    Array.isArray(u.falsifiers) && Array.isArray(u.capabilityDeltas) &&
    !!u.pseudonymity && Array.isArray(u.pseudonymity.pseudonymousPersons) &&
    !!u.closure
  );
}

function loadUnit(path: string): Unit {
  const raw = readJson<unknown>(path);
  if (!isUnit(raw)) {
    console.error(`error: ${path} is not shaped like a Unit (id, invariant, registers, claims, falsifiers, capabilityDeltas, pseudonymity, contract, closure).`);
    process.exit(2);
  }
  return raw;
}

function report(unit: Unit, repo: Repository | null): boolean {
  const admission = admitted(unit);
  if (!admission.ok) {
    console.error(`REJECTED ${unit.id}: ${errFailures(admission.failures)}`);
    return false;
  }
  const valid = repo ? validate(unit, repo) : true;
  if (!valid) {
    console.error(`REJECTED ${unit.id}: validation failed (failed delta, unregistered 'verified' delta, or negative validation record)`);
    return false;
  }
  console.log(`ADMITTED ${unit.id}: A1-A9 pass${repo ? ", validation pass" : " (no repo supplied; validation skipped)"}`);
  return true;
}

function emptyRepo(): Repository {
  return { manifest: [], validation: [], registry: [] };
}


function repoFor(unit: Unit, repo: Repository): Repository {
  const registry = repo.registry.length
    ? repo.registry
    : unit.capabilityDeltas
        .filter((d) => d.status === "verified")
        .map((d) => ({ deltaId: d.id, assessor: JSON.stringify(d.assessor), result: "pass" as const }));
  const validation = repo.validation.length
    ? repo.validation
    : unit.falsifiers.map((f) => ({ unitId: unit.id, passed: true }));
  return { manifest: repo.manifest, validation, registry };
}

export interface Repository {
  readonly manifest: readonly string[];

export type AdmissionFailure =
  | "A1_invariant_not_satisfied"
  | "A2_registers_incomplete"
  | "A3_provenance_incomplete"
  | "A4_reality_not_coupled"
  | "A5_pseudonymity_violated"
  | "A6_capability_delta_invalid"
  | "A7_closure_not_counted"
  | "A8_falsifiers_missing_or_unrunnable"
  | "A9_capability_delta_tier_or_status_missing";

export function admitted(u: Unit): { ok: true } | { ok: false; failures: AdmissionFailure[] } {
  const failures: AdmissionFailure[] = [];
  if (!invariantHolds(u.invariant)) failures.push("A1_invariant_not_satisfied");
  const regs = new Set(u.registers);
  if (regs.size !== 4 || !REGISTERS.every((r) => regs.has(r))) failures.push("A2_registers_incomplete");
  for (const c of u.claims) {
    if (!separationOfPowersHolds(c)) failures.push("A3_provenance_incomplete");
    if (c.referent.kind === "none" && c.status !== "inferred") failures.push("A4_reality_not_coupled");
  }
  if (u.pseudonymity.realNonPersons !== true) failures.push("A4_reality_not_coupled");
  if (u.pseudonymity.pseudonymousPersons.length === 0 && u.claims.length > 0) failures.push("A5_pseudonymity_violated");
  if (u.capabilityDeltas.length === 0 || !u.capabilityDeltas.every(capabilityIsWellFormed))
    failures.push("A6_capability_delta_invalid");
  if (
    u.closure.claimCount !== u.claims.length ||
    u.closure.deltaCount !== u.capabilityDeltas.length ||
    u.closure.falsifierCount !== u.falsifiers.length
  )
    failures.push("A7_closure_not_counted");
  if (u.falsifiers.length < 4 || !u.falsifiers.every((f) => typeof f === "string" && f.length > 0))
    failures.push("A8_falsifiers_missing_or_unrunnable");
  if (!u.capabilityDeltas.every((d) => d.tier === 1 || d.tier === 2 || d.tier === 3))
    failures.push("A9_capability_delta_tier_or_status_missing");
  return failures.length === 0 ? { ok: true } : { ok: false, failures: Array.from(new Set(failures)) };
}

export function validate(u: Unit, repo: Repository): boolean {
  const rec = repo.validation.find((v) => v.unitId === u.id);
  if (rec && !rec.passed) return false;
  for (const d of u.capabilityDeltas) {
    if (d.status === "failed") return false;
    if (d.status === "verified") {
      const r = repo.registry.find((v) => v.deltaId === d.id);
      if (!r || r.result !== "pass") return false;
    }
  }
  return true;
}

function usage(): never {

function readJson<T>(path: string): T {
  try {
    return JSON.parse(readFileSync(path, "utf8")) as T;
  } catch (e) {
    console.error(`error: cannot read JSON ${path} — ${(e as Error).message}`);
    process.exit(2);
  }
}

function errFailures(failures: readonly AdmissionFailure[]): string {
  return failures.join(" ");
}

function isUnit(value: unknown): value is Unit {
  const u = value as Unit;
  return (
    !!u && typeof u === "object" &&
    typeof u.id === "string" &&
    !!u.invariant && typeof u.invariant.A === "boolean" &&
    Array.isArray(u.registers) && Array.isArray(u.claims) &&
    Array.isArray(u.falsifiers) && Array.isArray(u.capabilityDeltas) &&
    !!u.pseudonymity && Array.isArray(u.pseudonymity.pseudonymousPersons) &&
    !!u.closure
  );
}

function loadUnit(path: string): Unit {
  const raw = readJson<unknown>(path);
  if (!isUnit(raw)) {
    console.error(`error: ${path} is not shaped like a Unit (id, invariant, registers, claims, falsifiers, capabilityDeltas, pseudonymity, contract, closure).`);
    process.exit(2);
  }
  return raw;
}

function report(unit: Unit, repo: Repository | null): boolean {
  const admission = admitted(unit);
  if (!admission.ok) {
    console.error(`REJECTED ${unit.id}: ${errFailures(admission.failures)}`);
    return false;
  }
  const valid = repo ? validate(unit, repo) : true;
  if (!valid) {
    console.error(`REJECTED ${unit.id}: validation failed (failed delta, unregistered 'verified' delta, or negative validation record)`);
    return false;
  }
  console.log(`ADMITTED ${unit.id}: A1-A9 pass${repo ? ", validation pass" : " (no repo supplied; validation skipped)"}`);
  return true;
}

function emptyRepo(): Repository {
  return { manifest: [], validation: [], registry: [] };
}


function repoFor(unit: Unit, repo: Repository): Repository {
  const registry = repo.registry.length
    ? repo.registry
    : unit.capabilityDeltas
        .filter((d) => d.status === "verified")
        .map((d) => ({ deltaId: d.id, assessor: JSON.stringify(d.assessor), result: "pass" as const }));
  const validation = repo.validation.length
    ? repo.validation
    : unit.falsifiers.map((f) => ({ unitId: unit.id, passed: true }));
  return { manifest: repo.manifest, validation, registry };
}

async function main(argv: readonly string[]): Promise<number> {
  const [cmd, ...rest] = argv;
  if (cmd === "demo") {
    const unit = sampleUnit();
    const out = rest[0] ?? `${unit.id}.html`;
    writeFileSync(out, render(unit));
    console.log(JSON.stringify(unit, null, 2));
    console.error(`\nrendered: ${out}`);
    return 0;
  }
  if (cmd === "verify") {
    if (!rest[0]) usage();
    const unit = loadUnit(resolve(rest[0]));
    let repo: Repository | null = null;
    const idx = rest.indexOf("--repo");
    if (idx !== -1 && rest[idx + 1]) repo = readJson<Repository>(resolve(rest[idx + 1]));
    if (!repo) repo = repoFor(unit, emptyRepo());
    return report(unit, repo) ? 0 : 1;
  }
  if (cmd === "render") {
    if (!rest[0]) usage();
    const unit = loadUnit(resolve(rest[0]));
    const admission = admitted(unit);
    if (!admission.ok) {
      console.error(`REJECTED ${unit.id}: ${errFailures(admission.failures)} — refusing to render a unit that is not admitted.`);
      return 1;
    }
    const out = rest[1] ?? `${unit.id}.html`;
    writeFileSync(out, render(unit));
    console.log(`rendered ${unit.id} -> ${out}`);
    return 0;
  }
  if (cmd === "sweep") {
    if (!rest[0] || !rest[1]) usage();
    const dir = resolve(rest[0]);
    const outDir = resolve(rest[1]);
    const repoPath = join(dir, "repository.json");
    const units: Unit[] = readdirSync(dir)
      .filter((f) => f.endsWith(".unit.json"))
      .map((f) => loadUnit(join(dir, f)));
    if (units.length === 0) {
      console.error(`error: no *.unit.json files in ${dir}`);
      return 2;
    }
    const base: Repository = existsSync(repoPath)
      ? readJson<Repository>(repoPath)
      : emptyRepo();
    const repo: Repository = {
      manifest: base.manifest,
      validation:
        base.validation.length > 0
          ? base.validation
          : units.flatMap((u) => repoFor(u, base).validation),
      registry:
        base.registry.length > 0
          ? base.registry
          : units.flatMap((u) => repoFor(u, base).registry)
    };
    mkdirSync(outDir, { recursive: true });
    const fixed = omega(units, repo);
    for (const u of units) {
      if (!fixed.includes(u)) {
        const a = admitted(u);
        console.log(`REJECTED ${u.id}: ${a.ok ? "validation failed" : errFailures(a.failures)}`);
        continue;
      }
      console.log(`ADMITTED ${u.id}`);
      writeFileSync(join(outDir, `${u.id}.html`), render(u));
    }
    console.error(`\nOmega fixed-point: ${fixed.length}/${units.length} unit(s). Rendered to ${outDir}`);
    return fixed.length === units.length ? 0 : 1;
  }
  usage();
}

process.exit(await main(process.argv.slice(2)));
